const baseJob = { title: 'Python', url: 'https://hh.ru/vacancy/1', company: 'Fixture', locationText: 'Москва', remote: true, salaryText: 'от 200 000 ₽ в месяц до вычета налогов', currency: 'RUB' };
function harness({ failPage, challenge = false, empty = false, changed = false, launchFailure = false, unknown = false, slowFirst = 0, job = baseJob } = {}) {
  let closed = false, contexts = 0, launches = 0, requests = 0;
  const urls = [];
  const makePage = () => {
    let index = 0;
    return {
      async goto(url) { index++; requests++; urls.push(url); if (slowFirst && index === 1) await new Promise(r => setTimeout(r, slowFirst)); if (index === failPage) { if (unknown) throw new Error('unexpected'); return { ok: () => false, status: () => 403 }; } return { ok: () => true }; },
      async waitForFunction() {}, async waitForTimeout() {}, async close() {},
      locator(selector) { return {
        async innerText() { return challenge ? 'captcha' : empty ? 'ничего не найдено' : 'Jobs'; },
        async count() { return selector.includes('pager-next') ? 1 : selector.includes('no-results') ? Number(empty) : 1; },
        async evaluateAll() { return empty || changed ? [] : [structuredClone(job)]; }
      }; }
    };
  };
  const context = { newPage: async () => makePage(), close: async () => {} };
  return { launch: async () => { launches++; if (launchFailure) throw new Error('bad launch'); return { newContext: async () => { contexts++; return context; }, close: async () => { closed = true; } }; },
    state: () => ({ closed, launches, contexts, requests, urls }) };
}

test('salary is the midpoint of a complete RUB monthly range; everything else stays only in compensation', async () => {
  const salaryOf = async (salaryText, currency = 'RUB') => (await searchHh({ query: 'Python', pages: 1, launch: harness({ job: { ...baseJob, salaryText, currency } }).launch })).jobs[0];
  const range = await salaryOf('200 000–350 000 ₽ в месяц на руки');

  assert.deepEqual([range.compensation.min, range.compensation.max, range.compensation.period, range.compensation.taxBasis], [200000, 350000, 'month', 'net']);
  assert.deepEqual((await salaryOf('100 000 ₽ в месяц')).salary, { min: 100000, max: 100000, currency: 'RUB' });
  for (const [text, currency] of [['от 200 000 ₽ в месяц', 'RUB'], ['до 200 000 ₽ в месяц', 'RUB'], ['2 000–3 000 $ в месяц', 'USD'], ['200 000–350 000 ₽ в год', 'RUB'], ['200 000–350 000 ₽ за смену', 'RUB'], ['200 000–350 000 ₽', 'RUB'], ['200 000–350 000 ₽ за неделю', 'RUB'], ['', 'RUB']]) {
    const job = await salaryOf(text, currency);
    assert(!('salary' in job), text);
    if (text) assert(job.compensation, text);
  }
  const [merged] = mergeJobs([{ ...baseJob, compensation: parseCompensation('200 000–350 000 ₽ в месяц') }, { ...baseJob, salary: { min: 1, max: 2, currency: 'RUB' } }]);
  assert.deepEqual(merged.salary, { min: 275000, max: 275000, currency: 'RUB' });
  assert.deepEqual([CONTRACT_VERSION, TOOL_VERSION], [3, '0.8.0']);

    }
  } finally { await browser.close(); }
});
