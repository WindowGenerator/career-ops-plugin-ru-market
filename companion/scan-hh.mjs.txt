const clean = s => typeof s === 'string' ? s.replace(/\s+/gu, ' ').trim() : '';
const diagnostic = (category, message, extra = {}) => Object.assign(new Error(message), { category, ...extra });
const sleep = ms => new Promise(r => setTimeout(r, ms));
const abortInfo = signal => signal?.reason === 'deadline'

const salaryFrom = c => c?.currency && (c.min !== undefined || c.max !== undefined)
  ? { ...(c.min !== undefined ? { min: c.min } : {}), ...(c.max !== undefined ? { max: c.max } : {}), currency: c.currency } : undefined;
// Same normalisation and flags as the plugin's makeJob; flagged jobs are kept.
export function sanitizeJob(job) {
  const flags = new Set(Array.isArray(job.injectionFlags) ? job.injectionFlags.filter(f => typeof f === 'string' && /^[a-z][a-z0-9-]{0,39}$/.test(f)) : []);
  const safe = (value, field) => { const result = untrusted(value, field); result.flags.forEach(flag => flags.add(flag)); return clean(result.text); };
  const out = { ...job, title: safe(job.title, 'title'), company: safe(job.company, 'company'), location: safe(job.location, 'location'),
    locationText: safe(job.locationText, 'location'), note: safe(job.note, 'note') };
  if (Array.isArray(job.locations)) out.locations = job.locations.map(x => safe(x, 'location')).filter(Boolean);
  if (flags.size) out.injectionFlags = [...flags].sort(); else delete out.injectionFlags;
  return out;
}
const atomicJson = (path, value) => { mkdirSync(dirname(path), { recursive: true }); writeFileSync(path + '.tmp', JSON.stringify(value, null, 2) + '\n'); renameSync(path + '.tmp', path); };
const queryId = (query, index = 0) => `q${String(index + 1).padStart(2, '0')}-${createHash('sha256').update(query).digest('hex').slice(0, 8)}`;

export async function extractHhJobs(page, identity = {}) {
  const raw = await page.locator('[data-qa="vacancy-serp__vacancy"]').evaluateAll(cards => {
    const text = (card, selector) => card.querySelector(selector)?.textContent?.replace(/\s+/gu, ' ').trim() || '';
    return cards.flatMap(card => {
      const link = card.querySelector('[data-qa="serp-item__title"], [data-qa="vacancy-serp__vacancy-title"]');
      if (!link) return [];
      let url;
      try { url = new URL(link.getAttribute('href'), document.baseURI); } catch { return []; }
      if (url.protocol !== 'https:' || url.username || url.password || url.port || url.hostname !== 'hh.ru' || !/^\/vacancy\/[1-9]\d*\/?$/.test(url.pathname)) return [];
      const title = link.textContent?.replace(/\s+/gu, ' ').trim();
      if (!title) return [];
      const currency = [...card.querySelectorAll('data[value]')].find(el => /^[A-Z]{3}$/.test(el.value));
      const salaryText = text(card, '[data-qa="vacancy-serp__vacancy-compensation"], [data-qa="vacancy-serp__compensation"]')
        || currency?.parentElement?.textContent?.replace(/\s+/gu, ' ').trim() || '';
      const remote = !!card.querySelector('[data-qa="vacancy-label-work-schedule-remote"]');
      return [{ title, url: `https://hh.ru${url.pathname.replace(/\/$/, '')}`,
        company: text(card, '[data-qa="vacancy-serp__vacancy-employer-text"], [data-qa="vacancy-serp__vacancy-employer"]'),
        locationText: text(card, '[data-qa="vacancy-serp__vacancy-address"]'), remote, salaryText, currency: currency?.value }];
    });
  });
  return raw.map(item => {
    const sourceId = /\/vacancy\/(\d+)/.exec(item.url)?.[1];
    const compensation = parseCompensation(item.salaryText, item.currency);
    const salary = salaryFrom(compensation);
    return sanitizeJob({ title: item.title, url: item.url, company: item.company, source: 'hh', transport: 'playwright', sourceId,
      locationText: item.locationText, locations: item.locationText ? [item.locationText] : [],
      location: [item.locationText, item.remote ? 'Можно удалённо' : ''].filter(Boolean).join(' / '),
      workArrangement: item.remote ? 'remote' : 'unknown', eligibility: { status: 'unknown' },
      dataLevel: 'listing', hasDescription: false, matchedQueries: identity.query ? [{ query: identity.query, queryId: identity.queryId }] : [],
      ...(compensation ? { compensation } : {}), ...(salary ? { salary } : {}),
      note: `source: hh; transport: playwright${item.salaryText ? '; salary: ' + item.salaryText : ''}` });
  }).filter(job => job.title);
}

export function mergeJobs(jobs) {
  const merged = new Map();
  for (const job of jobs) {
    const previous = merged.get(job.url);
    if (!previous) { merged.set(job.url, structuredClone(job)); continue; }
    const queries = new Map([...(previous.matchedQueries ?? []), ...(job.matchedQueries ?? [])].map(q => [q.queryId, q]));
    previous.matchedQueries = [...queries.values()];
    for (const key of ['title', 'company', 'locationText', 'compensation', 'workArrangement']) {
      const old = previous[key], next = job[key];
      const empty = value => value == null || value === '' || value === 'unknown';
      if (empty(old) && !empty(next)) { previous[key] = structuredClone(next); if (key === 'locationText') { previous.location = job.location; previous.locations = job.locations; } }
      else if (!empty(old) && !empty(next) && JSON.stringify(old) !== JSON.stringify(next)) {
        previous.provenance ??= { conflicts: [] };
        const conflict = { field: key, retained: old, observed: next, matchedQueries: job.matchedQueries ?? [] };
        if (!previous.provenance.conflicts.some(x => JSON.stringify(x) === JSON.stringify(conflict))) previous.provenance.conflicts.push(conflict);
      }
    }
  }
  return [...merged.values()].map(job => {
    const salary = salaryFrom(job.compensation);
    if (salary) job.salary = salary; else delete job.salary;
    const flags = [...new Set(jobs.filter(x => x.url === job.url).flatMap(x => x.injectionFlags ?? []))].sort();
    if (flags.length) job.injectionFlags = flags;
    return job;
  });
}

export function validateOptions(options) {
  if (!clean(options.query) || clean(options.query).length > 512) throw diagnostic('invalid-config', 'A query of 1..512 characters is required');
  if (!Number.isInteger(options.pages) || options.pages < 1 || options.pages > 20) throw diagnostic('invalid-config', 'pages must be an integer between 1 and 20');
  if (options.area !== undefined && options.area !== '' && !/^[1-9]\d*$/.test(String(options.area))) throw diagnostic('invalid-config', 'area must be an HH numeric area ID');
  if (!['chrome', 'chromium'].includes(options.channel)) throw diagnostic('invalid-config', 'channel must be chrome or chromium');
  if (!Number.isInteger(options.retries) || options.retries < 0 || options.retries > 2) throw diagnostic('invalid-config', 'retries must be 0..2');
  if (typeof options.titleOnly !== 'boolean' || typeof options.headed !== 'boolean') throw diagnostic('invalid-config', 'titleOnly and headed must be booleans');
  if (options.searchField !== undefined && (!Array.isArray(options.searchField) || options.searchField.some(x => !SEARCH_FIELDS.includes(x)))) throw diagnostic('invalid-config', `search_field must be a list of ${SEARCH_FIELDS.join(', ')}`);
  if (options.excludedText !== undefined && (typeof options.excludedText !== 'string' || !clean(options.excludedText) || clean(options.excludedText).length > 512)) throw diagnostic('invalid-config', 'excluded_text must be 1..512 characters');
  if (options.professionalRole !== undefined && (!Array.isArray(options.professionalRole) || options.professionalRole.length > 20 || options.professionalRole.some(x => !/^[1-9]\d*$/.test(String(x))))) throw diagnostic('invalid-config', 'professional_role must be a list of up to 20 numeric HH role IDs');
  if (options.onlyWithSalary !== undefined && typeof options.onlyWithSalary !== 'boolean') throw diagnostic('invalid-config', 'only_with_salary must be boolean');
  if (options.timeoutMs !== undefined && (!Number.isInteger(options.timeoutMs) || options.timeoutMs < 1000 || options.timeoutMs > 3_600_000)) throw diagnostic('invalid-config', 'timeout_ms must be an integer from 1000 to 3600000');
}
const SEARCH_FIELDS = ['name', 'company_name', 'description'];
export async function preflight({ channel = 'chrome', headed = false, launch } = {}) {
  if (!['chrome', 'chromium'].includes(channel) || typeof headed !== 'boolean') throw diagnostic('invalid-config', 'Invalid browser channel or headed flag');
  let chromium;
  if (!launch) {
    try { ({ chromium } = await import('playwright')); }
    catch { throw diagnostic('browser-launch', 'Playwright is missing in career-ops; install career-ops dependencies.'); }
    launch = options => chromium.launch(options);
  }
  try { return await launch({ headless: !headed, ...(channel === 'chromium' ? {} : { channel }) }); }
  catch (error) {
    const message = /Executable doesn't exist|not found|distribution.*not found/i.test(error.message)
      ? `Selected browser ${channel} is missing; install Chrome or career-ops Playwright Chromium.`
      : `Browser launch failed (${channel}); check environment/sandbox permissions. No sandbox settings were changed.`;
    throw diagnostic('browser-launch', message);
  }
}
function classify(error) {
  return { category: error.category ?? (/Timeout|timeout|net::|ECONN|ENOTFOUND/i.test(`${error.name} ${error.message}`) ? 'network' : 'unknown'),
    message: clean(error.category ? error.message : `${error.name || 'Error'} while collecting HH listings`).slice(0, 240),
    ...(Number.isInteger(error.http_status) ? { http_status: error.http_status } : {}) };
}
const statusBase = options => ({ source: 'hh', transport: 'playwright', query: options.query, queryId: options.queryId ?? queryId(clean(options.query)), completed_pages: 0, count: 0 });
const envelope = (jobs, statuses, extra = {}) => ({ contractVersion: CONTRACT_VERSION, toolVersion: TOOL_VERSION, jobs, sourceStatuses: statuses, ...extra });

export async function searchHh(input) {
  const options = { pages: 1, channel: 'chrome', titleOnly: false, headed: false, retries: 0, ...input };
  const status = statusBase(options);
  let browser, context, page, ownBrowser = false;
  let jobs = input.initial?.jobs ?? [], start = input.initial?.completed_pages ?? 0;
  status.completed_pages = start;
  try {
    validateOptions(options);
    browser = input.browser ?? await preflight(options); ownBrowser = !input.browser;
    context = input.context ?? (browser.newContext ? await browser.newContext({ locale: 'ru-RU', serviceWorkers: 'block' }) : browser);
    page = await context.newPage();
    for (let index = start; index < options.pages; index++) {
      if (options.signal?.aborted) { const info = abortInfo(options.signal); throw diagnostic(info.category, info.message); }
      const url = new URL('https://hh.ru/search/vacancy');
      url.searchParams.set('text', options.query); url.searchParams.set('page', String(index));
      if (options.area) url.searchParams.set('area', String(options.area));
      for (const field of new Set([...(options.titleOnly ? ['name'] : []), ...(options.searchField ?? [])])) url.searchParams.append('search_field', field);
      if (options.excludedText) url.searchParams.set('excluded_text', clean(options.excludedText));
      for (const role of options.professionalRole ?? []) url.searchParams.append('professional_role', String(role));
      if (options.onlyWithSalary) url.searchParams.set('only_with_salary', 'true');
      let found, hasNext;
      for (let attempt = 0; ; attempt++) {
        try {
          const response = await page.goto(url.href, { waitUntil: 'domcontentloaded', timeout: 45000 });
          if (!response?.ok()) {
            const code = response?.status();
            throw diagnostic([401, 403].includes(code) ? 'access' : 'http', `HH search HTTP ${code ?? 'unknown'}`, { http_status: code });
          }
          try {
            await page.waitForFunction(() => document.querySelector('[data-qa="vacancy-serp__vacancy"], [data-qa="vacancy-serp__no-results"], [data-qa="vacancy-serp__empty"]')
              || /ничего не найдено|вакансий не найдено|проверка.*человек|подтвердите.*человек|captcha|access denied/i.test(document.body.innerText), null, { timeout: 15000 });
          } catch (error) {
            if (error.name === 'TimeoutError') throw diagnostic('broken-markup', 'HH card, empty or challenge marker absent');
            throw error;
          }
          const body = await page.locator('body').innerText();
          if (/captcha|access denied|проверка.*человек|подтвердите.*человек/i.test(body)) throw diagnostic('access', 'HH access challenge; collection stopped');
          found = await extractHhJobs(page, { query: options.query, queryId: status.queryId });
          const explicitEmpty = /ничего не найдено|вакансий не найдено/i.test(body)
            || await page.locator('[data-qa="vacancy-serp__no-results"], [data-qa="vacancy-serp__empty"]').count() > 0;
          if (!found.length && !explicitEmpty) throw diagnostic('broken-markup', 'HH markup changed: no valid vacancy cards');
          hasNext = found.length > 0 && await page.locator('[data-qa="pager-next"]').count() > 0;
          break;
        } catch (error) {
          if (classify(error).category !== 'network' || attempt >= options.retries) throw error;
          await sleep(500 * (attempt + 1));
        }
      }
      jobs = mergeJobs([...jobs, ...found]); status.completed_pages++; status.count = jobs.length;
      status.status = 'ok'; status.stop_reason = hasNext ? 'page-limit' : found.length ? 'exhausted' : 'empty';
      await input.onCheckpoint?.({ jobs, ...status });
      console.error(`HH ${status.queryId} page ${index + 1}: ${found.length} vacancies`);
      if (!hasNext) break;
      if (index + 1 < options.pages) await (page.waitForTimeout?.(1000) ?? sleep(1000));
    }
    status.status = 'ok'; status.stop_reason ??= 'page-limit';
  } catch (error) {
    Object.assign(status, { status: status.completed_pages ? 'partial' : 'failed', failed_page: status.completed_pages + 1, ...classify(error) });
  } finally {
    await page?.close?.().catch(() => {});
    if (!input.context && context && context !== browser) await context.close().catch(() => {});
    if (ownBrowser) await browser?.close().catch(() => {});
  }
  status.count = jobs.length;
  const flagged = jobs.filter(job => job.injectionFlags?.length).length;
  if (flagged) status.injection_flagged = flagged;
  return envelope(jobs, [status]);
}

export async function loadBatchConfig(path, entryName, dataRoot = getCareerOpsRoot()) {
  const { load } = await import('js-yaml');
  const file = resolve(dataRoot, path);
  let data;
  try { data = load(readFileSync(file, 'utf8')); } catch { throw diagnostic('invalid-config', 'Cannot read or parse selected YAML config'); }
  if (!data || typeof data !== 'object') throw diagnostic('invalid-config', 'Config must be an object');
  const scannerConfig = Object.fromEntries(Object.entries(data).filter(([key]) => !['hh_browser', 'job_boards', 'tracked_companies'].includes(key)));
  const block = data.hh_browser ?? (() => {
    const candidates = [...(data.job_boards ?? []), ...(data.tracked_companies ?? [])].filter(entry => entry?.hh_browser && (!entryName || entry.name === entryName));
    if (candidates.length !== 1) throw diagnostic('invalid-config', 'Select one entry containing hh_browser with --entry, or provide a top-level hh_browser block');
    return candidates[0].hh_browser;
  })();
  return { ...blockOptions(block), scannerConfig, configFile: file };
}

const BLOCK_KEYS = new Set(['queries', 'pages', 'area', 'titleOnly', 'channel', 'headed', 'retries', 'artifactDir', 'search_field', 'excluded_text', 'professional_role', 'only_with_salary', 'timeout_ms']);
function blockOptions(block) {
  if (!block || typeof block !== 'object' || Array.isArray(block)) throw diagnostic('invalid-config', 'hh_browser must be an object');
  for (const key of Object.keys(block)) if (!BLOCK_KEYS.has(key)) throw diagnostic('invalid-config', `Unsupported hh_browser option: ${key}`);
  const { search_field, excluded_text, professional_role, only_with_salary, timeout_ms, ...rest } = block;
  const list = value => (Array.isArray(value) ? value : [value]).map(String);
  return { ...rest, ...(search_field !== undefined ? { searchField: list(search_field) } : {}),
    ...(excluded_text !== undefined ? { excludedText: excluded_text } : {}), ...(professional_role !== undefined ? { professionalRole: list(professional_role) } : {}),
    ...(only_with_salary !== undefined ? { onlyWithSalary: only_with_salary } : {}), ...(timeout_ms !== undefined ? { timeoutMs: timeout_ms } : {}) };
}

export async function collectBatch(input) {
  const dataRoot = input.dataRoot ?? getCareerOpsRoot();
  const options = { pages: 1, channel: 'chrome', titleOnly: false, headed: false, retries: 0, ...input };
  if (!Array.isArray(options.queries) || !options.queries.length || options.queries.length > 100
    || options.queries.some(q => typeof q !== 'string' || !clean(q))) throw diagnostic('invalid-config', 'queries must contain 1..100 non-empty strings');
  options.queries = [...new Set(options.queries.map(clean))];
  options.queries.forEach(query => validateOptions({ ...options, query }));
  const folder = resolve(dataRoot, options.artifactDir ?? `data/scan-output/hh-${new Date().toISOString().slice(0, 10)}`);
  const checkpointPath = resolve(folder, 'checkpoint.json');
  const parameters = { queries: options.queries, pages: options.pages, area: options.area ?? '', titleOnly: options.titleOnly,
    channel: options.channel, headed: options.headed, retries: options.retries, searchField: [...(options.searchField ?? [])].sort(),
    excludedText: options.excludedText ?? '', professionalRole: [...(options.professionalRole ?? [])].map(String).sort(), onlyWithSalary: options.onlyWithSalary ?? false };
  const signature = createHash('sha256').update(JSON.stringify(parameters)).digest('hex');
  let checkpoint = { contractVersion: CONTRACT_VERSION, toolVersion: TOOL_VERSION, signature, parameters, queries: {} };
  if (existsSync(checkpointPath)) {
    if (!options.resume) throw diagnostic('invalid-config', 'Artifacts already exist; choose a new folder or use --resume');
    checkpoint = JSON.parse(readFileSync(checkpointPath, 'utf8'));
    if (checkpoint.signature !== signature || checkpoint.contractVersion !== CONTRACT_VERSION) throw diagnostic('invalid-config', 'Resume parameters or contract version do not match checkpoint');
  } else if (options.resume) throw diagnostic('invalid-config', 'No checkpoint to resume');
  let browser, context, launchFailure, blocked = false;
  const results = [];
  const save = () => atomicJson(checkpointPath, checkpoint);
  save();
  try {
    const pending = options.queries.some((query, i) => checkpoint.queries[queryId(query, i)]?.status !== 'ok');
    if (pending) {
      try { browser = await preflight(options); context = browser.newContext ? await browser.newContext({ locale: 'ru-RU', serviceWorkers: 'block' }) : browser; }
      catch (error) { launchFailure = classify(error); }
    }
    for (const [index, query] of options.queries.entries()) {
      const id = queryId(query, index), old = checkpoint.queries[id];
      let result;
      if (old?.status === 'ok') result = envelope(old.jobs, [{ ...old, jobs: undefined }]);
      else if (blocked || options.signal?.aborted || launchFailure) {
        const stop = abortInfo(options.signal);
        const state = { ...statusBase({ query, queryId: id }), completed_pages: old?.completed_pages ?? 0, count: old?.jobs?.length ?? 0, status: launchFailure && !results.length ? 'failed' : 'skipped', category: blocked ? 'access' : options.signal?.aborted ? stop.category : launchFailure.category,
          message: blocked ? 'Skipped after HH access block' : options.signal?.aborted ? stop.message : launchFailure.message };
        result = envelope(old?.jobs ?? [], [state]);
      } else {
        result = await searchHh({ ...options, query, queryId: id, browser, context, initial: old,
          onCheckpoint: state => { checkpoint.queries[id] = { ...state, status: state.completed_pages >= options.pages || state.stop_reason !== 'page-limit' ? 'ok' : 'running' }; save(); } });
      }
      const status = result.sourceStatuses[0];
      checkpoint.queries[id] = { ...status, jobs: result.jobs }; save();
      atomicJson(resolve(folder, `query-${String(index + 1).padStart(2, '0')}.json`), result);
      results.push(result);
      if (status.category === 'access') blocked = true;
      if (index + 1 < options.queries.length && !blocked && browser) await (options.wait ?? sleep)(1000);
    }
  } finally {
    if (context && context !== browser) await context.close().catch(() => {});
    await browser?.close().catch(() => {});
  }
  const result = envelope(mergeJobs(results.flatMap(r => r.jobs)), results.flatMap(r => r.sourceStatuses), { parameters, artifactDir: folder });
  atomicJson(resolve(folder, 'merged.json'), result);
  atomicJson(resolve(folder, 'queries-summary.json'), { ...result, jobs: undefined });
  // Same executable imports the cache; no generated JavaScript adapters.
  const portals = { ...options.scannerConfig, tracked_companies: [], job_boards: [{ name: 'HH browser cache', provider: 'local-parser', careers_url: 'https://hh.ru/search/vacancy',
    parser: { command: 'node', script: 'scripts/ru-market/scan-hh.mjs', args: ['--import-cache', resolve(folder, 'merged.json')], timeout_ms: 20000, max_buffer_bytes: 10000000 } }] };
  const { dump } = await import('js-yaml');
  writeFileSync(resolve(folder, 'portals-hh.yml'), dump(portals));
  if (options.scan) {
    const coreRoot = resolve(dirname(fileURLToPath(import.meta.url)), '../..');
    const scanArgs = ['scan.mjs', '--json', ...(options.dryRun ? ['--dry-run'] : [])];
    let stdout, stderr;
    try { ({ stdout, stderr } = await promisify(execFile)(process.execPath, scanArgs, { cwd: coreRoot,
      env: { ...process.env, CAREER_OPS_PORTALS: resolve(folder, 'portals-hh.yml'), CAREER_OPS_ROOT: dataRoot }, timeout: 120000, maxBuffer: 10000000 })); }
    catch (error) { if (!error.stdout) throw diagnostic('scan-import', 'Core scan failed; inspect configuration and rerun cache import'); ({ stdout, stderr } = error); }
    writeFileSync(resolve(folder, 'scan.log'), stdout + (stderr ?? ''));
    try { result.scanReceipt = JSON.parse(stdout); }
    catch { throw diagnostic('scan-import', 'Core did not return a JSON scan receipt; see scan.log'); }
    atomicJson(resolve(folder, 'scan-receipt.json'), result.scanReceipt);
  }
  return result;
}

export function readCache(path) {
  const data = JSON.parse(readFileSync(path, 'utf8'));
  if (data.contractVersion !== CONTRACT_VERSION || !Array.isArray(data.jobs) || !Array.isArray(data.sourceStatuses)) throw diagnostic('invalid-config', 'Invalid HH cache contract');
  const valid = data.jobs.every(j => j.source === 'hh' && typeof j.title === 'string' && /^https:\/\/hh\.ru\/vacancy\/[1-9]\d*$/.test(j.url));
  if (!valid) throw diagnostic('invalid-config', 'Cache contains invalid HH job URLs or metadata');
  return { ...data, jobs: data.jobs.map(sanitizeJob).filter(job => job.title) };
}

export async function main(args = process.argv.slice(2), deps = {}) {
  const options = {}, write = deps.write ?? (text => process.stdout.write(text));
  if (args.includes('--help')) {
    write('HH tool ' + TOOL_VERSION + ': --query TEXT | --config FILE [--entry NAME] | --import-cache FILE | --preflight; [--pages 1..20] [--area ID] [--title-only] [--search-field name,company_name,description] [--excluded-text TEXT] [--professional-role ID,ID] [--only-with-salary] [--timeout-ms MS] [--channel chrome|chromium] [--artifact-dir DIR | --no-artifacts] [--resume] [--retries 0..2] [--output FILE] [--scan [--dry-run]]\n'); return { exitCode: 0 };
  }
  if (args.includes('--version')) { write(TOOL_VERSION + '\n'); return { exitCode: 0 }; }
  const flags = { '--query': 'query', '--config': 'config', '--entry': 'entry', '--pages': 'pages', '--area': 'area', '--channel': 'channel',
    '--artifact-dir': 'artifactDir', '--output': 'output', '--import-cache': 'cache', '--retries': 'retries', '--search-field': 'searchField',
    '--excluded-text': 'excludedText', '--professional-role': 'professionalRole', '--timeout-ms': 'timeoutMs' };
  const switches = { '--title-only': 'titleOnly', '--headed': 'headed', '--resume': 'resume', '--preflight': 'preflight', '--scan': 'scan', '--dry-run': 'dryRun',
    '--only-with-salary': 'onlyWithSalary', '--no-artifacts': 'noArtifacts' };
  for (let i = 0; i < args.length; i++) {
    const key = args[i];
    if (switches[key]) options[switches[key]] = true;
    else if (flags[key]) { if (!args[i + 1] || args[i + 1].startsWith('--')) throw diagnostic('invalid-config', key + ' requires a value'); options[flags[key]] = args[++i]; }
    else throw diagnostic('invalid-config', 'Unknown option: ' + key);
  }
  for (const key of ['pages', 'retries', 'timeoutMs']) if (options[key] !== undefined) options[key] = Number(options[key]);
  for (const key of ['searchField', 'professionalRole']) if (options[key] !== undefined) options[key] = options[key].split(',').map(x => x.trim()).filter(Boolean);
  const controller = new AbortController();
  const abort = () => controller.abort();
  process.once('SIGINT', abort); process.once('SIGTERM', abort);
  let timer, temp;
  const arm = ms => { if (!timer && Number.isInteger(ms) && ms >= 1000) timer = setTimeout(() => controller.abort('deadline'), ms); };
  arm(options.timeoutMs);
  try {
    if (options.scan && !options.config) throw diagnostic('invalid-config', '--scan requires --config');
    if (options.noArtifacts && (options.scan || options.artifactDir || options.resume || !options.config)) throw diagnostic('invalid-config', '--no-artifacts only combines with --config and no --scan, --artifact-dir or --resume');
    if ([options.query, options.config, options.cache, options.preflight].filter(Boolean).length > 1) throw diagnostic('invalid-config', 'Select one query/config/cache/preflight route');
    const common = { launch: deps.launch, signal: controller.signal };
    let result;
    if (options.cache) result = readCache(resolve(getCareerOpsRoot(), options.cache));
    else if (options.preflight) { const browser = await preflight({ ...options, launch: deps.launch }); await browser.close(); result = envelope([], [], { preflight: 'ok' }); }
    else if (options.config) {
      const config = await loadBatchConfig(options.config, options.entry);
      if (options.noArtifacts) { temp = mkdtempSync(resolve(tmpdir(), 'hh-run-')); options.artifactDir = temp; }
      const merged = { ...config, ...options, ...common };
      arm(merged.timeoutMs);
      result = await collectBatch(merged);
      if (temp) delete result.artifactDir;
    } else result = await searchHh({ ...options, ...common });
    if (options.output) atomicJson(resolve(getCareerOpsRoot(), options.output), result);
    write(JSON.stringify(result, null, 2) + '\n');
    // Failures are reported through sourceStatuses with exit 0 so core can import partial jobs; only --scan and configuration errors are nonzero.
    const failed = result.sourceStatuses.some(s => s.status !== 'ok');
    return { result, exitCode: (options.scan && failed) || result.sourceStatuses.some(s => s.category === 'invalid-config') ? 1 : 0 };
  } finally {
    clearTimeout(timer); process.removeListener('SIGINT', abort); process.removeListener('SIGTERM', abort);
    if (temp) rmSync(temp, { recursive: true, force: true });
  }
}
if (isMainModule(import.meta.url)) main().then(({ exitCode }) => { process.exitCode = exitCode; }, error => {
  process.stdout.write(JSON.stringify(envelope([], [{ source: 'hh', transport: 'playwright', status: 'failed', completed_pages: 0, count: 0, ...classify(error) }])) + '\n');
  process.exitCode = 1;
});
